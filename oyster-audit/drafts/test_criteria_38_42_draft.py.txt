"""Tests for bin/lint_v3_prd_grounded.py — criteria #38-42 (Stream rc18).

Covers:
  #38 audio stream exists in mp4 + AAC codec (use ffprobe)
  #39 input-to-frame latency measurement file exists + <= 20ms
  #40 WASD balance — total_keyboard_events split (each direction within 25% +/- 15%)
  #41 stationary % <= 10% (no 2+ second windows where mouse_dx/dy=0 AND no keys pressed)
  #42 clip has < 2 sec frozen frame (consecutive identical frames)
"""
from __future__ import annotations

import importlib.util
import json
import sys
import textwrap
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parent.parent
LINT_MOD = REPO_ROOT / "bin" / "lint_v3_prd_grounded.py"

# Load lint_v3_prd_grounded as a module.
_spec = importlib.util.spec_from_file_location("lint_v3", LINT_MOD)
lint_v3 = importlib.util.module_from_spec(_spec)
sys.modules["lint_v3"] = lint_v3
_spec.loader.exec_module(lint_v3)

LintReport = lint_v3.LintReport
LintResult = lint_v3.LintResult


# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------

def _make_report(data_dir: Path) -> LintReport:
    """Run all checks and return the report."""
    return lint_v3.run_all_checks(data_dir)


def _result_by_id(rpt: LintReport, cid: int) -> LintResult | None:
    for r in rpt.results:
        if r.criterion_id == cid:
            return r
    return None


def _write_json(d: Path, name: str, data):
    p = d / name
    p.write_text(json.dumps(data), encoding="utf-8")
    return p


def _write_text(d: Path, name: str, text: str):
    p = d / name
    p.write_text(text, encoding="utf-8")
    return p


def _make_action_camera(d: Path, frames: list) -> Path:
    return _write_json(d, "action_camera.json", frames)


def _make_dummy_mp4(d: Path, name: str = "video.mp4") -> Path:
    """Create a minimal valid MP4 file (just the ftyp box)."""
    # Minimal ftyp box: size(4) + 'ftyp'(4) + brand(4) + minor(4) + compat(4)
    data = (
        b'\x00\x00\x00\x1c'  # size = 28
        b'ftyp'
        b'mp42'              # major brand
        b'\x00\x00\x00\x00'  # minor version
        b'mp42'              # compatible brand
        b'isom'              # compatible brand
    )
    p = d / name
    p.write_bytes(data)
    return p

        # 100 frames, 25 each of W, A, S, D = 25% each
        frames = []
        for i in range(100):
            key = ["W", "A", "S", "D"][i % 4]
            frames.append({

                "mouse_dx": 1.0,

# Criterion 42: Frozen frame < 2 sec
# ---------------------------------------------------------------------------

class TestCriterion42FrozenFrames:
    """#42: clip has < 2 sec frozen frame (consecutive identical frames)."""

    def test_no_video_fails(self, tmp_path):
        rpt = _make_report(tmp_path)
        r = _result_by_id(rpt, 42)
        assert r is not None
        assert r.passed is False
        assert "No primary video" in r.message

    def test_fallback_no_frozen_passes(self, tmp_path):
        # All frames have different mouse_dx -> no frozen
        frames = []
        for i in range(100):
            frames.append({
                "frame_index": i,
                "mouse_dx": float(i),
                "mouse_dy": 0.0,
                "camera_rotation_quaternion": [0.0, 0.0, 0.0, 1.0],
                "fps": 30.0,
            })
        _make_action_camera(tmp_path, frames)
        rpt = _make_report(tmp_path)
        r = _result_by_id(rpt, 42)
        assert r is not None
        # Without ffmpeg, falls back to action_camera check
        # All frames differ -> 0 frozen -> pass
        assert r.passed is True

    def test_fallback_frozen_fails(self, tmp_path):
        # 100 frames, all identical -> 99 consecutive frozen at 30fps = 3.3s
        frames = []
        for i in range(100):
            frames.append({
                "frame_index": i,
                "mouse_dx": 0.0,
                "mouse_dy": 0.0,
                "camera_rotation_quaternion": [0.0, 0.0, 0.0, 1.0],
                "fps": 30.0,
            })
        _make_action_camera(tmp_path, frames)
        rpt = _make_report(tmp_path)
        r = _result_by_id(rpt, 42)
        assert r is not None
        # Without ffmpeg, falls back -> 99 frozen frames at 30fps = 3.3s > 2s
        assert r.passed is False

    def test_fallback_short_frozen_passes(self, tmp_path):
        # 100 frames, 30 consecutive identical = 1.0s at 30fps (< 2s)
        frames = []
        for i in range(30):
            frames.append({
                "frame_index": i,
                "mouse_dx": 0.0,
                "mouse_dy": 0.0,
                "camera_rotation_quaternion": [0.0, 0.0, 0.0, 1.0],
                "fps": 30.0,
            })
        for i in range(30, 100):
            frames.append({
                "frame_index": i,
                "mouse_dx": float(i),
                "mouse_dy": 0.0,
                "camera_rotation_quaternion": [0.0, 0.0, 0.0, 1.0],
                "fps": 30.0,
            })
        _make_action_camera(tmp_path, frames)
        rpt = _make_report(tmp_path)
        r = _result_by_id(rpt, 42)
        assert r is not None
        assert r.passed is True
